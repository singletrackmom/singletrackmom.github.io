% Carlito (Calibri-metric) throughout, body and headings.
\usepackage{fontspec}
\setmainfont{Carlito}

% Ragged right (left-aligned, never justified).
\usepackage[document]{ragged2e}

% Block paragraphs: no first-line indent, small space between paragraphs.
\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt plus 2pt}

% Cover graphic support. build.sh puts the airc-sss dir on TEXINPUTS so
% cover-notitle.png resolves from cover.tex.
\usepackage{graphicx}

% One inch margins.
\usepackage[margin=1in]{geometry}

% Footer page numbers centered; no header rule. Cover page is set
% unnumbered from cover.tex via \thispagestyle{empty}.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\cfoot{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% All headings BLACK (explicit, so it never drifts to blue).
\usepackage{xcolor}
\usepackage{sectsty}
\allsectionsfont{\color{black}}
