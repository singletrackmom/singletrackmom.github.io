def short(s,adult=False):
    l=s.lower()
    if 'not a gemini step' in l: return 'Outside Gemini'
    if 'not tested' in l or 'not run' in l: return 'Not tested'
    if 'tool not offered' in l: return 'Not offered'
    if 'feature blocked; content completed' in l: return 'Tool missing, chat worked'
    if 'feature blocked; content blocked' in l: return 'Refused'
    if 'no report' in l: return 'No report'
    if 'not completed' in l: return 'Not completed'
    if 'failed' in l: return 'Failed'
    if 'feature blocked; content partial' in l or 'partial' in l: return 'Partial'
    if 'blocked' in l: return 'Refused'
    if 'deep research' in l: return 'Done with Deep Research'
    if 'completed' in l: return 'Done'
    return s

def pages(rows,budget=4300):
    out=[];cur=[];n=0
    for r in rows:
        L=len(r[2])+120
        if cur and n+L>budget: out.append(cur);cur=[];n=0
        cur.append(r);n+=L
    if cur: out.append(cur)
    return out
